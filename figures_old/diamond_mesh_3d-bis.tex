\tdplotsetmaincoords{50}{-20}

\begin{tikzpicture}[
    ddfv,
    scale=3,
    tdplot_main_coords
]

\def\Ax{0}    
\def\Ay{0}    
\def\Az{0}   

\def\Bx{1}    
\def\By{-0.4} 
\def\Bz{0.4} 

\def\Cx{2}  
\def\Cy{0}   
\def\Cz{0}   

\def\Dx{1}  
\def\Dy{0.4} 
\def\Dz{0.4} 

\coordinate (A) at (\Ax,\Ay,\Az);
\coordinate (B) at (\Bx,\By,\Bz);
\coordinate (C) at (\Cx,\Cy,\Cz);
\coordinate (D) at (\Dx,\Dy,\Dz);

% Projections / quadrilatère supérieur
\def\Apx{\Ax}
\def\Apy{\Ay}
\def\Apz{\Bz}

\def\Bpx{\Bx}
\def\Bpy{\By}
\def\Bpz{\Bz}

\def\Cpx{\Cx}
\def\Cpy{\Cy}
\def\Cpz{\Bz}

\def\Dpx{\Dx}
\def\Dpy{\Dy}
\def\Dpz{\Dz}

\coordinate (Ap) at (\Apx,\Apy,\Apz);
\coordinate (Bp) at (\Bpx,\Bpy,\Bpz);
\coordinate (Cp) at (\Cpx,\Cpy,\Cpz);
\coordinate (Dp) at (\Dpx,\Dpy,\Dpz);

% Quadrilatère A-D-Cp-Bp
\draw[
    fill=gray,
    fill opacity=0.25,
    draw=black,
    thick
]
(Ap) -- (Dp) -- (Cp) -- (Bp) -- cycle;

% Bord de la surface réglée
\draw[thick] (A) -- (B) -- (C) -- (D) -- cycle;

% Génératrices AB -> DC
\foreach \s in {0,0.1,...,1}{
    \coordinate (P) at (
        {(1-\s)*\Ax + \s*\Bx},
        {(1-\s)*\Ay + \s*\By},
        {(1-\s)*\Az + \s*\Bz}
    );

    \coordinate (Q) at (
        {(1-\s)*\Dx + \s*\Cx},
        {(1-\s)*\Dy + \s*\Cy},
        {(1-\s)*\Dz + \s*\Cz}
    );

    \draw[thin] (P) -- (Q);
}

% Génératrices AD -> BC
\foreach \s in {0,0.1,...,1}{
    \coordinate (P) at (
        {(1-\s)*\Ax + \s*\Dx},
        {(1-\s)*\Ay + \s*\Dy},
        {(1-\s)*\Az + \s*\Dz}
    );

    \coordinate (Q) at (
        {(1-\s)*\Bx + \s*\Cx},
        {(1-\s)*\By + \s*\Cy},
        {(1-\s)*\Bz + \s*\Cz}
    );

    \draw[thin] (P) -- (Q);
}

% Courbe rouge
\draw[
    thick,
    red,
    variable=\t,
    domain=0:1,
    samples=20
]
plot (
{
(1-\t)*((1-\t)*\Ax + \t*\Bx)
+
\t*((1-\t)*\Dx + \t*\Cx)
},
{
(1-\t)*((1-\t)*\Ay + \t*\By)
+
\t*((1-\t)*\Dy + \t*\Cy)
},
{
(1-\t)*((1-\t)*\Az + \t*\Bz)
+
\t*((1-\t)*\Dz + \t*\Cz)
}
);

% Courbe bleue
\draw[
    thick,
    blue,
    variable=\t,
    domain=0:1,
    samples=20
]
plot (
{
\t*((1-\t)*\Ax + \t*\Bx)
+
(1-\t)*((1-\t)*\Dx + \t*\Cx)
},
{
\t*((1-\t)*\Ay + \t*\By)
+
(1-\t)*((1-\t)*\Dy + \t*\Cy)
},
{
\t*((1-\t)*\Az + \t*\Bz)
+
(1-\t)*((1-\t)*\Dz + \t*\Cz)
}
);

\def\lam{0.3}

% Flèche lambda
\draw[->, thick]
(A) -- (
    {\Bx+\lam*(\Bx-\Ax)},
    {\By+\lam*(\By-\Ay)},
    {\Bz+\lam*(\Bz-\Az)}
)
node[pos=1,right] {\contour{white}{$\lambda$}};

% Flèche mu
\draw[->, thick]
(A) -- (
    {\Dx+\lam*(\Dx-\Ax)},
    {\Dy+\lam*(\Dy-\Ay)},
    {\Dz+\lam*(\Dz-\Az)}
)
node[pos=1,right] {$\mu$};

% Labels
\node[below] at (A) {$X_1$};
\node[below] at (B) {$B_1$};
\node[right] at (C) {$X_2$};
\node[left] at (D) {$B_2$};

% Diagonales
\draw[<->] (A) -- (C) node[midway] {\contour{white}{$u$}};
\draw[<->] (B) -- (D) node[midway] {\contour{white}{$v$}};

\end{tikzpicture}

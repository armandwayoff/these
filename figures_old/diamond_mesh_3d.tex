\newcommand{\ruledsurface}[5][0.1]{%
    % #1 = pas, par défaut 0.1
    % #2,#3,#4,#5 = A,B,C,D
    \foreach \t in {0,#1,...,1}{%
        % Génératrices AB -> DC
        \draw[gray, thin, opacity=0.5] ($(#2)!\t!(#3)$) -- ($(#5)!\t!(#4)$);

        % Génératrices AD -> BC
        \draw[gray, thin, opacity=0.5] ($(#2)!\t!(#5)$) -- ($(#3)!\t!(#4)$);
    }%
}

\chapter{Inégalités de type \nompropre{Poincaré}}

\section{Inégalité de \nompropre{Poincaré}}

\begin{lemme}
Soit $\Omega$ un ouvert de $\R^N$ \textbf{borné} dans au moins une direction de l'espace. Il existe une constante $C > 0$ telle que, pour toute fonction $v \in \H^1_0(\Omega)$,
\begin{equation}\label{eq:inegalite_de_poincare}
\int_\Omega \abs{v(x)}^2 \d x \leqslant C \int_\Omega \abs{\grad v(x)}^2 \d x.
\end{equation}
\end{lemme}

\begin{demo}
Par un argument standart de densité, il suffit de démontrer \ref{eq:inegalite_de_poincare} pour une fonction régulière $v~\in~\Cont^1(\overline{\Omega})$ s'annulant sur le bord $\partial \Omega$. On peut l'étendre par continuité par zéro en dehors de $\Omega$. L'hypothèse sur le caractère borné de $\Omega$ dit (après une éventuelle rotation) que pour tout $x \in \Omega$, la première composante $x_1$ est bornée, $-\infty < a \leqslant x_1 \leqslant b < \infty$. Pour tout $x \in \Omega$, on a
\[
v(x) = \int_a^{x_1} \pdv{v}{x_1}\,(t, x_2, \dots, x_N) \d t,
\]
d'où l'on déduit par l'inégalité de \nompropre{Cauchy}--\nompropre{Schwarz}
\[
\abs{v(x)}^2 \leqslant (b - a) \int_a^b \abs{\pdv{v}{x_1}\,(t, x_2, \dots, x_N)}^2 \d t.
\]
Intégrant sur $\Omega$, on obtient
\[
\int_\Omega \abs{v(x)}^2 \d x \leqslant (b - a) \int_\Omega \int_a^b \abs{\pdv{v}{x_1}\,(t, x_2, \dots, x_N)}^2 \d t \d x,
\]
et l'intégrande étant positive et indépendante de $x_1$, on peut permuter les deux intégrations par rapport à $t$ et $x_1$, et conclure
\begin{align*}
\int_\Omega \abs{v(x)}^2 \d x &\leqslant (b - a) \int_a^b \d x_1 \int_\Omega \abs{\pdv{v}{x_1}\,(t, x_2, \dots, x_N)}^2 \d t \d x_2 \cdots \d x_N \\
&\leqslant (b - a)^2 \int_\Omega \abs{\grad v(x)}^2 \d x.
\end{align*}
On remarque au passage que la constante $C$ dans l'inégalité de \nompropre{Poincaré} \ref{eq:inegalite_de_poincare} est plus petite que le carré du diamètre de $\Omega$.
\end{demo}

\begin{lemme}\label{lemme:pw_convexe}
Soit $\Omega$ un ouvert borné, régulier et \textbf{convexe} de $\R^N$. On désigne par $V$ l'espace
\[
V = \ens[\Big]{\phi \in \H^1(\Omega) \tq \int_\Omega \phi(x) \d x = 0}.
\]
Il existe une constante $0 < C \leqslant 2^{N-1} \, d(\Omega)^2$ telle que, pour toute fonction $v \in V$,
\begin{equation}\label{eq:inegalite_de_poincare_wirtinger}
\int_\Omega \abs{v(x)}^2 \d x \leqslant C \int_\Omega \abs{\grad v(x)}^2 \d x.
\end{equation}
\end{lemme}

\begin{demo}
Encore une fois, par un argument standard de densité, il suffit de démontrer \ref{eq:inegalite_de_poincare_wirtinger} pour une fonction régulière $v \in V \cap \Cont^1(\overline{\Omega})$. L'idée ingénieuse de \nompropre{Poincaré} est de remarquer tout d'abord que, pour tout $v \in V$, on a
\begin{equation}\label{eq:8}
\int_\Omega \int_\Omega \big(v(x) - v(x')\big)^2 \d x \d x' = 2 \abs{\Omega} \int_\Omega v(x)^2 \d x
\end{equation}
puisque $v$ est à moyenne nulle sur $\Omega$. Ensuite, grâce à la convexité de $\Omega$, il remarque que, pour tout couple de points $x, x' \in \Omega$, le segment $(x\,, x')$ est inclus dans $\Omega$ et on a
\[
v(x) - v(x') = \int_0^1 (x - x') \cdot \grad v \big(t\,x + (1-t)\,x'\big) \d t.
\]
En élevant au carré, l'inégalité de \nompropre{Cauchy}--\nompropre{Schwarz} conduit à
\[
\abs{v(x) - v(x')}^2 \leqslant d(\Omega)^2 \int_0^1 \abs{\grad v \big(t\,x + (1-t)\,x'\big)}^2 \d t,
\]
où $d(\Omega) \defeq \sup_{x, x' \in \Omega} \abs{x - x'}$ désigne le diamètre de $\Omega$. On en déduit alors
\[
\int_\Omega \int_\Omega \abs{v(x) - v(x')}^2 \d x \d x'
\leqslant d(\Omega)^2 \int_\Omega \int_\Omega \left( \int_0^{1/2} + \int_{1/2}^1 \right) \abs{\grad v \big(t\,x + (1-t)\,x'\big)}^2 \d t \d x \d x'.
\]
Par symétrie on a
\[
\int_\Omega \int_\Omega \int_0^{1/2} \abs{\grad v \big(t\,x + (1-t)\,x'\big)}^2 \d t \d x \d x'
 = \int_\Omega \int_\Omega \int_{1/2}^1 \abs{\grad v \big(t\,x + (1-t)\,x'\big)}^2 \d t \d x \d x',
\]
et donc
\[
\int_\Omega \int_\Omega \abs{v(x) - v(x')}^2 \d x \d x' \leqslant 2 \, d(\Omega)^2 \int_\Omega \int_\Omega \int_{1/2}^1 \abs{\grad v \big(t\,x + (1-t)\,x'\big)}^2 \d t \d x \d x'.
\]
Par \nompropre{Fubini}, on peut échanger l'ordre des intégrales en $t$ et $x$.  Pour $t$ fixé dans $\intervalleff{1/2}{1}$, le changement de variables $\tilde{x} = t\,x + (1-t)\, x'$ conduit à
\[
\int_\Omega \abs{\grad v \big(t\,x + (1-t)\,x'\big)}^2 \d x = \frac{1}{t^N} \int_\omega \abs{\grad v(\tilde{x})}^2 \d \tilde{x}
\]
où $\omega = t\, \Omega + (1 - t) \, x'$ est un sous-ensemble de $\Omega$ (car $\Omega$ est convexe). On peut donc majorer, pour $t \geqslant 1/2$,
\[
\int_\Omega \abs{\grad v \big(t\,x + (1-t)\,x'\big)}^2 \d x
\leqslant
2^N \int_\omega \abs{\grad v(\tilde{x})}^2 \d \tilde{x}
\leqslant
2^N \int_\Omega \abs{\grad v(\tilde{x})}^2 \d \tilde{x}.
\]
Au total, on a obtenu
\begin{align*}
\int_\Omega \int_\Omega \abs{v(x) - v(x')}^2 \d x \d x'
&\leqslant 2 \, d(\Omega)^2 \, 2^N \int_{1/2}^1 \int_\Omega \int_\Omega \abs{\grad v(x)}^2 \d x \d x' \d t \\
&\leqslant 2^N \, d(\Omega)^2 \, \abs{\Omega} \int_\Omega \abs{\grad v(x)}^2 \d x,
\end{align*}
ce qui conduit, par combinaison avec \ref{eq:8}, à l'inégalité de \nompropre{Poincaré}--\nompropre{Wirtinger} \ref{eq:inegalite_de_poincare_wirtinger}.
\end{demo}

La démonstration usuelle de l'inégalité de \nompropre{Poincaré}--\nompropre{Wirtinger} \ref{eq:inegalite_de_poincare_wirtinger} est valable pour tout ouvert borné \emph{connexe} (mais pas nécessairement convexe). Mais elle n'est pas constructive (au sens où elle ne donne aucune estimation sur la taille de la constante $C$) et repose sur un \textbf{argument de compacité}. Décrivons brièvement cette preuve par contradiction. Supposons que l'inégalité \ref{eq:inegalite_de_poincare_wirtinger} soit fausse : il existe donc une suite de fonctions $v_n \in V$ telle que
\[
\int_\Omega \abs{v_n(x)}^2 \d x = 1
\quad \text{et} \quad
\int_\Omega \abs{\grad v_n(x)}^2 \d x = 1/n.
\]
La suite $v_n$ étant bornée dans $\H^1(\Omega)$, par application du \textbf{théorème de compacité de \nompropre{Rellich}}, on peut en extraire une sous-suite $v_{n'}$ qui converge (fortement) vers une limite $v_\infty$ dans $\L^2(\Omega)$. Comme $\grad v_n$ converge manifestement vers $0$ dans $\L^2(\Omega)^N$, on en déduit que $\grad v_\infty = 0$. En effet, pour toute fonction test $\varphi \in \Distrib(\Omega)$ et pour tout $i \in \intervalleentier{1}{N}$, on a
\[
\int_\Omega \upPartial_{x_i} v_\infty(x) \, \varphi(x) \d x = 
- \int_\Omega v_\infty(x) \, \upPartial_{x_i} \varphi(x) \d x 
= - \lim_{n' \to \infty} \int_\Omega v_{n'}(x) \, \upPartial_{x_i} \varphi(x) \d x 
= \lim_{n' \to \infty}\int_\Omega \upPartial_{x_i} v_{n'}(x) \,\varphi(x) \d x = 0.
\]
De plus, $\Omega$ étant connexe, la fonction $v_\infty$ est constante. Puisque $v_n \in V$ et que $V$ est par construction un sous-espace fermé de $\H^1(\Omega)$, la fonction $v_\infty$ appartient à l'espace $V$ et est donc nulle. Ceci est une contradiction avec le fait que (continuité de la norme)
\[
\int_\Omega \abs{v_\infty(x)}^2 \d x = \lim_{n' \to \infty} \int_\Omega \abs{v_{n'}(x)}^2 \d x = 1.
\]

\begin{lemme}[Généralisation du lemme \ref{lemme:pw_convexe}]
Soit $\Omega$ un ouvert borné, régulier et simplement connexe de $\R^N$. On désigne par $W$ l'espace
\[
W = \ens[\Big]{\phi \in \H^1(\Omega) \tq \int_{\partial \Omega} \phi(s) \d s = 0}.
\]
Il existe une constante $C > 0$ telle que, pour toute fonction $v \in W$,
\[
\int_{\partial \Omega} \abs{v(s)}^2 \d s \leqslant C \int_\Omega \abs{\grad v(x)}^2 \d x.
\]
\end{lemme}

\begin{remarque}[Rapport \textsc{pr}e]
Remarquons qu'obtenir une inégalité locale de type \nompropre{Poincaré}, c'est-à-dire estimer la quantité
\[
\norme{v - \fint_B v(x) \d x}^2_{\L^2(B)} = \int_B \abs{\fint_B \big(v(y) - v(x)\big) \d x}^2 \d y \leqslant \int_B \fint_B \abs{v(y) - v(x)}^2 \d x \d y
\]
revient à estimer la différence $\abs{v(y) - v(x)}^2$. 
\end{remarque}

\section{Démonstration de l'inégalité de \nompropre{Poincaré}--\nompropre{Wirtinger}}\label{annexeWirtinger}

\begin{theo}[\nompropre{Rellich}]
    Soit $\Omega$ un ouvert borné de $\R^d$ de classe $\Cont^1$. Alors, de toute suite bornée de $\H^1(\Omega)$, on peut extraire une sous-suite qui converge dans $\L^2(\Omega)$. 
\end{theo}

\begin{theo}[Inégalité de \nompropre{Poincaré}--\nompropre{Wirtinger}, {[p.36]{ann201}}]
    Soit $\Omega$ un ouvert borné et connexe. Alors, il existe une constante $C > 0$ telle que pour tout $v \in \H^1(\Omega)$,
    \[
    \norme{v}^2_{\L^2(\Omega)} \leqslant C \left( \norme{\grad v}^2_{\L^2(\Omega)} + \frac{1}{\abs{\Omega}} \left( \int_\Omega v(x) \d x \right)^2 \right).
    \]
\end{theo}

\begin{lemme}
    On considère un ouvert $\Omega \subset \R^d$ connexe, borné et régulier de classe $\Cont^1$. Soit $\Phi$ une forme linéaire continue sur $\H^1(\Omega)$ telle que $\Phi(1_\Omega) \not= 0$, où $1_\Omega$ est la fonction constante égale à $1$ sur $\Omega$. Alors, il existe une constante $C_\Omega$ telle que pour tout $v \in \H^1(\Omega)$,
    \[
    \norme{v}_{\L^2(\Omega)} \leqslant C_\Omega \big( \norme{\grad v}_{\L^2(\Omega)} + \abs{\Phi(v)} \big). 
    \]
\end{lemme}

\begin{demo}[source:{\url{http://www.cmap.polytechnique.fr/~antonin/TDs2014/pc4.pdf}, p.5}]
On raisonne par l'absurde et on suppose donc que
\[
\forall n \in \Ne, \quad \exists v_n \in \H^1(\Omega), \quad \norme{v_n}_ {\L^2(\Omega)} > n \big( \norme{\grad v_n}_{\L^2(\Omega)} + \abs{\Phi(v_n)} \big).
\]
On peut normaliser $v_n$ par sa norme $\L^2(\Omega)$ (puisqu'elle n'est pas nulle), on a donc que pour tout $n \in \Ne$, il existe une suite $(v_n)_{n \in \Ne} \in \H^1(\Omega)$ telle que
\[
\begin{cases}
    \norme{v_n}_{\L^2(\Omega)} = 1,\\
    \norme{\grad v_n}_{\L^2(\Omega)} + \abs{\Phi(v_n)} < \frac{1}{n}.
\end{cases}
\]
La suite $(v_n)_{n \in \Ne}$ est bornée dans $\H^1(\Omega)$ et comme $\Omega$ est borné et régulier, on peut appliquer le théorème de \nompropre{Rellich}. On peut donc en extraire une sous-suite notée $(z_n)_{n \in \Ne}$ qui converge (fortement) au sens de la norme $\L^2(\Omega)$ vers $z \in \L^2(\Omega)$. La norme étant une fonction continue de $\L^2(\Omega)$ dans $\R$, on a:
\[
\norme{z}_{\L^2(\Omega)} = 1.
\]
Par ailleurs, de $\norme{\grad v_n}_{\L^2(\Omega)} < \frac{1}{n}$, on en déduit que la suite $(\grad z_n)_{n \in \Ne}$ converge fortement (au sens de la norme $\L^2$) vers $0 \in \L^2(\Omega)^d$. La suite $(z_n)_{n \in \Ne}$ est donc de \nompropre{Cauchy} dans $\H^1(\Omega)$ et donc converge vers $w \in \H^1(\Omega)$, puisque $\H^1(\Omega)$ est complet. Par unicité de la limite dans $\L^2(\Omega)$, on a $z = w$ dans $\L^2(\Omega)$. Soit $\varphi \in \Distrib(\Omega)$ une fonction test, on a pour tout $1 \leqslant i \leqslant d$,
\[
\int_\Omega z_n(x) \, \frac{\uppartial \varphi}{\uppartial x_i}(x) \d x = - \int_\Omega \frac{\uppartial z_n}{\uppartial x_i}(x) \, \varphi(x) \d x
\]
soit, en passant à la limite $n \to \infty$ et en rebasculant la dérivée partielle sur $z$,
\[
\int_\Omega \frac{\uppartial z}{\uppartial x_i}(x) \, \varphi(x) \d x = 0.
\]
On en déduit que la $i$-ème dérivée partielle faible de $z$ est égale à $0$ au sens des distributions et comme ces dérivées partielles sont dans $\L^2(\Omega)$, puisque $z \in \H^1(\Omega)$, on a finalement que $\grad z = 0$ presque partout. De plus, comme $\Omega$ est connexe, la fonction $z$ est constante égale à $Z$ sur $\Omega$. Montrons que $Z = 0$. De $\abs{\Phi(v_n)} < \frac{1}{n}$, en utilisant la continuité et la linéarité de $\Phi$ de $\H^1(\Omega)$ dans $\R$, il vient $\Phi(z) = Z \, \Phi(1_\Omega) = 0$. Sous l'hypothèse, indispensable, $\Phi(1_\Omega) \not= 0$, il vient $z = Z = 0$, ce qui contredit $\norme{z}_{\L^2(\Omega)} = 1$. 
\end{demo}

\begin{demo}[de l'inégalité de \nompropre{Poincaré}--\nompropre{Wirtinger}]
On considère l'application 
\[
\fonction[\Phi]{\H^1(\R)}{\R}{v}{\frac{1}{\sqrt{\abs{\Omega}}} \int_\Omega v(x) \d x}.
\]
Il s'agit bien d'une forme linéaire continue sur $\H^1(\Omega)$ puisque par l'inégalité de \nompropre{Cauchy}--\nompropre{Schwarz}, pour tout $v \in \H^1(\Omega)$, 
\[
\abs{\frac{1}{\sqrt{\abs{\Omega}}} \int_\Omega v(x) \d x} \leqslant \frac{1}{\sqrt{\abs{\Omega}}} \times \sqrt{\abs{\Omega}} \norme{v}_{\L^2(\Omega)} \leqslant \norme{v}_{\H^1(\Omega)}.
\]
De plus, $\Phi(1_\Omega) = \sqrt{\abs{\Omega}} \not= 0$. Nous pouvons donc appliquer le lemme, selon lequel il existe $C_\Omega$ tel que pour tout $v \in \H^1(\Omega)$,
\[
\norme{v}_{\L^2(\Omega)} \leqslant C_\Omega \left( \norme{\grad v}_{\L^2(\Omega)} + \frac{1}{\sqrt{\abs{\Omega}}} \int_\Omega v(x) \d x \right).
\]
En élevant au carré, on obtient finalement le résultat
\[
\norme{v}^2_{\L^2(\Omega)} \leqslant C \left( \norme{\grad v}^2_{\L^2(\Omega)} + \frac{1}{\abs{\Omega}} \left( \int_\Omega v(x) \d x\right)^2 \right)
\]
avec $C = 2 \, C_\Omega$.
\end{demo}


\section{Valeurs propres et fonctions propres du laplacien}

La motivation de \nompropre{Poincaré} pour démontrer le lemme \ref{lemme:pw_convexe} est l'étude des valeurs propres et des fonctions propres du laplacien, solutions de
\begin{equation}
    \begin{cases}
    \begin{aligned}
        - \laplacien u_j &= k_j \, u_j &&\text{dans } \Omega \\
        \upPartial_n u_j + h \, u_j &= 0 &&\text{sur } \partial \Omega \\
    \end{aligned}
    \end{cases}
\end{equation}
où $k_j \in \R$ est une valeurs propre et $u_j(x)$ une fonction propre non nulle. Le paramètre $h$ appartient à $\Rp$ et les deux valeurs extrêmes $h = 0$, $h = +\infty$ permettent de retrouver les conditions aux limites de \nompropre{Neumann} ou de \nompropre{Dirichlet} respectivement. 

\begin{mdframed}
L'inégalité de \nompropre{Poincaré} n'est rien d'autre que l'énoncé que la première valeur propre du laplacien est strictement positive.
\end{mdframed}
La meilleure constante de \nompropre{Poincaré} est l'inverse de la première valeur propre du laplacien.

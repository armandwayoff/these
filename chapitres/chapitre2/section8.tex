\section{Main theorem}

Cette section est consacrée à la démonstration de l'inégalité de Poincaré \ddfv sur une surface fermée sans bord de $\R^3$ énoncée par le théorème suivant. 

\begin{theo} \label{theo:main}
    Soit $\ensddfvplat$ un maillage \ddfv de la surface polyédrique $\surfpoly$. Il existe une constante $C$, indépendante de la taille du maillage $\size(\ensddfvplat)$, telle que pour tout $u^\ensddfv \in \Rddfvmn$, on a
    \begin{equation} \label{eq:main_ddfv}
    \normeddfvT{u^\ensddfv} \leqslant C \normeddfvD{\grad^\ensdiamant u^\ensddfv}.
    \end{equation}
    L'ensemble $\Rddfvmn$ défini par \Cref{eq:moyenne_nulle} est l'ensemble des fonctions scalaires \ddfv à moyenne nulle. 
\end{theo}

\subsection{Une inégalité de graphe}

\begin{defi} \label{defi:graphe_ddfv}
Soit $\ensddfvplat= (\ensprimalplat, \ensdualplat)$ un maillage \ddfv plat. On lui associe deux graphes:
\begin{itemize}
    \item le \defini{graphe du maillage primal} $\graphe^\ensprimalplat = (\enssomdualplat, \ensaretedualplat)$ où $\enssomdualplat$ est l'ensemble des sommets de $\ensdualplat$ et $\ensaretedualplat$ est l'ensemble des arêtes de $\ensdualplat$;
    \item le \defini{graphe du maillage dual} $\graphe^\ensdualplat = (\enssomprimplat, \ensareteprimplat)$ où $\enssomprimplat$ est l'ensemble des sommets de $\ensprimalplat$ et $\ensareteprimplat$ est l'ensemble des arêtes de $\ensprimalplat$.
\end{itemize}
\end{defi}

\begin{envide}[Notations]
On note $\vertices(\graphe)$ l'ensemble des sommets d'un graphe $\graphe$ et $\edges(\graphe)$ l'ensemble de ses arêtes. On pourra également utiliser la notation $\vertices$ et $\edges$ lorsqu'il n'y a pas d'ambiguïté sur le graphe. 

Soit $u^\ensddfv = \big( u^\ensprimal, u^\ensdual \big) \in \R^\ensddfv$. On considère la fonction $u^\ensprimal$ comme une fonction sur la graphe $\graphe^\ensprimal$ dans le sens où pour $x_{\primal{K}} \in \enssomdual$, on note $u_{\primal{K}} = u^\ensprimal(x_{\primal{K}})$ et on considère la fonction $u^\ensdual$ comme une fonction sur la graphe $\graphe^\ensdual$ dans le sens où pour $x_{\mailleduale{K}} \in \enssomprim$, on note $u_{\mailleduale{K}} = u^\ensdual(x_{\mailleduale{K}})$.
\end{envide}

\noindent Pour tout $u^\ensddfv \in \R^\ensddfv$, on pose
\[
\mathcal{E}(u^\ensddfv) \defeq 
\sum_{\primal{K} \sim \primal{L}} ( u_{\primal{K}} - u_{\primal{L}} )^2
+
\sum_{\mailleduale{K} \sim \mailleduale{L}} ( u_{\mailleduale{K}} - u_{\mailleduale{L}} )^2.
\]
Pour obtenir \Cref{eq:main_ddfv}, l'objectif va être d'obtenir d'une part $\mathcal{E}(u^\ensddfv) \gtrsim \normeddfvT{u^\ensddfv}^2$ et d'autre part $\mathcal{E}(u^\ensddfv) \lesssim \normeddfvD{\grad^\ensdiamant u^\ensddfv}^2$. 

\begin{lemme}[Une inégalité de graphe] \label{lemme:ineg_de_graphe}
Pour tout $u^\ensddfv \in \R^\ensddfv$,
\[
\mathcal{E}(u^\ensddfv) \geqslant
\frac{h(\graphe^\ensddfvplat)^2}{2} \mathopen{}\left(
    \sum_{\primal{K} \in \ensprimal} \big( u_{\primal{K}} - \overline{u}^\ensprimal \big)^2 \degr(x_{\primal{K}})
    +
    \sum_{\mailleduale{K} \in \ensdual} \big( u_{\mailleduale{K}} - \overline{u}^\ensdual \big)^2 \degr(x_{\mailleduale{K}})
\right),
\]
où
\[
\overline{u}^\ensprimal = \frac{
    \sum\limits_{\primal{K} \in \ensprimal} u_{\primal{K}} \degr(x_{\primal{K}})
}{
    \sum\limits_{\primal{K} \in \ensprimal} \degr(x_{\primal{K}})
},
\quad
\overline{u}^\ensdual = \frac{
    \sum\limits_{\mailleduale{K} \in \ensdual} u_{\mailleduale{K}} \degr(x_{\mailleduale{K}})
}{
    \sum\limits_{\mailleduale{K} \in \ensdual} \degr(x_{\mailleduale{K}})
},
\quad
\text{et}
\quad
h(\graphe^\ensddfvplat) \defeq \min\mathopen{}\big\{ h(\graphe^\ensprimal), h(\graphe^\ensdual)\big\}.
\]
La constante $h(\graphe^\ensprimal)$ (resp. $h(\graphe^\ensdual)$) est la constante de Cheeger du graphe $\graphe^\ensprimal$ (resp. $\graphe^\ensdual$) définie par \Cref{eq:cheeger_constant}.
\end{lemme}

\begin{demo}
Soit $u^\ensddfv \in \R^\ensddfv$. On considère cette fonction comme une fonction sur les graphes $\graphe^\ensprimal$ et $\graphe^\ensdual$ comme expliqué dans les notations. Il suffit alors d'appliquer le \Cref{theo:cheeger_inequality}, établissant l'inégalité de Cheeger, d'une part au graphe $\graphe^\ensprimal$: 
\begin{equation} \label{eq:ineg_de_graphe_primal}
\sum_{\primal{K} \sim \primal{L}} ( u_{\primal{K}} - u_{\primal{L}} )^2
\geqslant 
\frac{h(\graphe^\ensprimal)^2}{2}
\sum_{\primal{K} \in \ensprimal} \big( u_{\primal{K}} - \overline{u}^\ensprimal \big)^2 \degr(x_{\primal{K}}),
\end{equation}
et d'autre part au graphe $\graphe^\ensdual$:
\begin{equation} \label{eq:ineg_de_graphe_dual}
\sum_{\mailleduale{K} \sim \mailleduale{L}} ( u_{\mailleduale{K}} - u_{\mailleduale{L}} )^2
\geqslant 
\frac{h(\graphe^\ensdual)^2}{2}
\sum_{\mailleduale{K} \in \ensdual} \big( u_{\mailleduale{K}} - \overline{u}^\ensdual \big)^2 \degr(x_{\mailleduale{K}}),
\end{equation}
où on a utilisé l'expression \Cref{eq:spectral_gap} pour les \emph{spectal gaps}. La somme des inégalités \ref{eq:ineg_de_graphe_primal} et \ref{eq:ineg_de_graphe_dual} donne le résultat. 
\end{demo}

\begin{lemme} \label{lemme:minoration_degre}
Soient $\primal{K} \in \ensprimal$ et $\mailleduale{K} \in \ensdual$. Alors
\[
\degr(x_{\primal{K}}) \geqslant \frac{\mesure(\primalplat{K})}{\reg(\ensddfvplat)^2 \size(\ensddfvplat)^2},
\quad
\degr(x_{\mailleduale{K}}) \geqslant \frac{\mesure(\mailledualeplate{K})}{\reg(\ensddfvplat)^2 \size(\ensddfvplat)^2}.
\]
\end{lemme}

\begin{demo}
On raisonne sur une maille primale plate $\primalplat{K}$. Posons $n_{\primal{K}} \defeq \degr(x_{\primal{K}})$. Chaque arête de $\primalplat{K}$ correspond à une maille primale voisine donc $n_{\primal{K}}$ est le nombre d'arêtes de $\primalplat{K}$. Puisque la maille est étoilée par rapport à $x_{\primalplat{K}}$, on peut la découper en $n_{\primal{K}}$ triangles d'intérieurs disjoints comme illustré sur la \Cref{fig:decoupage_primal_plat} :
\[
\primalplat{K}
=
\bigcup_{i = 1}^{n_{\primal{K}}}
\Delta(x_{\primalplat{K}}, s^\ast_i, s^\ast_{i+1}),
\]
où $(s^\ast_i)_{1 \leqslant i \leqslant n_{\primal{K}}}$ sont les sommets de la maille, ordonnés cycliquement avec la convention $s^\ast_{n_{\primal{K}} + 1} = s^\ast_1$. Si on note $\angle{i}^\ast$ l'angle $\widehat{s^\ast_i \, x_{\primalplat{K}} \, s^\ast_{i+1}}$, alors
\[
\mesure(\primalplat{K})
= 
\frac{1}{2} \sum_{i=1}^{n_{\primal{K}}}
\underbrace{\norme{s^\ast_i - x_{\primalplat{K}}}}_{\leqslant \diam{\primalplat{K}}} \underbrace{\norme{s^\ast_{i+1} - x_{\primalplat{K}}}}_{\leqslant \diam{\primalplat{K}}} \abs{\sin \angle{i}^\ast}
\leqslant
\frac{1}{2} \, n_{\primal{K}} \diam{\primalplat{K}}{}^2,
\] 
et on obtient en particulier
\begin{equation} \label{eq:estim}
\degr(x_{\primal{K}}) \geqslant \frac{\mesure(\primalplat{K})}{\diam{\primalplat{K}}{}^2}.
\end{equation}
Le même raisonnement s'applique aux mailles duales plates. On utilise cette fois le découpage illustré sur la \Cref{fig:decoupage_dual_plat} pour une maille $\mailledualeplate{K}$ :
\[
\mailledualeplate{K} = \bigcup_{i=1}^{n_{\mailleduale{K}}} \big[ 
    \Delta(x_{\mailleduale{K}}, s_i, m_i) \cup \Delta(x_{\mailleduale{K}}, m_i, s_{i+1})
    \big]
\]
où $n_{\mailleduale{K}} = \degr(x_{\mailleduale{K}})$, $(s_i)_{1 \leqslant i \leqslant n_{\mailleduale{K}}}$ sont les sommets de la maille, $(m_i)_{1 \leqslant i \leqslant n_{\mailleduale{K}}}$ sont les milieux des arêtes primales coupant la maille $\mailledualeplate{K}$ et $\angle{i} = \widehat{s_i \, x_{\mailleduale{K}} \, m_i}$, $\angle{i}' = \widehat{m_i \, x_{\mailleduale{K}} \, s_{i+1}}$. Alors,
\[
\mesure(\mailledualeplate{K})
=
\sum_{i=1}^{n_{\mailleduale{K}}}
\frac{1}{2}
\Big[
    \norme{s_i - x_{\mailleduale{K}}} \norme{m_i - x_{\mailleduale{K}}} \abs{\sin \theta_i}
    +
    \norme{m_i - x_{\mailleduale{K}}} \norme{s_{i+1} - x_{\mailleduale{K}}} \abs{\sin \theta'_i}
\Big]
\leqslant
n_{\mailleduale{K}} \diam{\mailleduale{K}}{}^2,
\]
et on obtient
\begin{equation} \label{eq:estim_ter}
\degr(x_{\mailleduale{K}}) \geqslant \frac{\mesure(\mailledualeplate{K})}{\diam{\mailledualeplate{K}}{}^2}.
\end{equation}
Dans la régularité $\reg(\ensddfvplat)$, les termes $\displaystyle\max_{\primalplat{K} \in \ensprimalplat} \max_{\diamantplat{D} \in \ensdiamantplat_{\primalplat{K}}} \frac{\diam{\primalplat{K}}}{\diam{\diamantplat{D}}}$ et $\displaystyle\max_{\mailledualeplate{K} \in \ensdualplat} \max_{\diamantplat{D} \in \ensdiamantplat_{\mailledualeplate{K}}} \frac{\diam{\mailledualeplate{K}}}{\diam{\diamantplat{D}}}$
permettent d'avoir les estimations
\begin{equation} \label{eq:estim_bis}
\diam{\primalplat{K}} \leqslant \reg(\ensddfvplat) \diam{\diamantplat{D}} \leqslant \reg(\ensddfvplat) \size(\ensddfvplat)
\quad
\text{et}
\quad
\diam{\mailledualeplate{K}} \leqslant \reg(\ensddfvplat) \diam{\diamantplat{D}} \leqslant \reg(\ensddfvplat) \size(\ensddfvplat).
\end{equation}
La combinaison de \Cref{eq:estim,eq:estim_ter,eq:estim_bis} donne le résultat.

\begin{remarque}
Avec les poids $\poids{\primal{K}} = \displaystyle\frac{\mesure(\primalplat{K})}{2}$ et $\poids{\mailleduale{K}} = \displaystyle\frac{\mesure(\mailledualeplate{K})}{2}$, les minorations deviennent
\[
\degr(x_{\primal{K}}) \geqslant \frac{2}{\reg(\ensddfvplat)^2 \size(\ensddfvplat)^2} \, \poids{\primal{K}},
\quad
\degr(x_{\mailleduale{K}}) \geqslant \frac{2}{\reg(\ensddfvplat)^2 \size(\ensddfvplat)^2} \, \poids{\mailleduale{K}}.
\]
\end{remarque}
Les \Cref{lemme:ineg_de_graphe,lemme:minoration_degre} vont nous permettre de montrer le résultat suivant. 

\begin{prop} \label{prop:minoration_energie}
Pour tout $u^\ensddfv \in \Rddfvmn$,
\[
\mathcal{E}(u^\ensddfv) \geqslant \frac{h(\graphe^\ensddfvplat)^2}{\reg(\ensddfvplat)^2 \size(\ensddfvplat)^2} \normeddfvT{u^\ensddfv}^2.
\]
L'ensemble $\Rddfvmn$ défini par \Cref{eq:moyenne_nulle} est l'ensemble des fonctions scalaires \ddfv à moyenne nulle. 
\end{prop}

\begin{demo}
Soit $u^\ensddfv \in \Rddfvmn$. Par définition, la moyenne géométrique $\sum\limits_{\primal{K} \in \ensprimal} \poids{\primal{K}} \, u_{\primal{K}}$ est nulle. De plus, elle minimise la fonction $c \mapsto \sum\limits_{\primal{K} \in \ensprimal} \poids{\primal{K}} \, (u_{\primal{K}} - c)^2$. Par conséquent,
\[
\sum_{\primal{K} \in \ensprimal} \poids{\primal{K}} \big( u_{\primal{K}} - \overline{u}^\ensprimal)^ 2
\geqslant
\sum_{\primal{K} \in \ensprimal} \poids{\primal{K}} \, u_{\primal{K}}{}^2,
\]
et en utilisant les estimations du \Cref{lemme:minoration_degre} on trouve
\begin{equation} \label{eq:ineg1}
\sum_{\primal{K} \in \ensprimal} \big(u_{\primal{K}} - \overline{u}^\ensprimal \big)^2 \degr(x_{\primal{K}})
\geqslant
\frac{2}{\reg(\ensddfvplat)^2 \size(\ensddfvplat)^2}
\sum_{\primal{K} \in \ensprimal} \poids{\primal{K}} \, u_{\primal{K}}{}^2.
\end{equation}
Le même raisonnement sur le maillage dual donne
\begin{equation} \label{eq:ineg2}
\sum_{\mailleduale{K} \in \ensdual} \big(u_{\mailleduale{K}} - \overline{u}^\ensdual \big)^2 \degr(x_{\mailleduale{K}})
\geqslant
\frac{2}{\reg(\ensddfvplat)^2 \size(\ensddfvplat)^2}
\sum_{\mailleduale{K} \in \ensdual} \poids{\mailleduale{K}} \, u_{\mailleduale{K}}{}^2.
\end{equation}
On somme les inégalités \ref{eq:ineg1} et \ref{eq:ineg2} puis le \Cref{lemme:ineg_de_graphe} et la définition de la norme $\normeddfvT{}$ donnent le résultat. 
\end{demo}

\noindent Intéressons-nous maintenant à la majoration de $\mathcal{E}$.

\begin{prop} \label{prop:majoration_energie}
Pour tout $u^\ensddfv \in \R^\ensddfv$,
\[
\mathcal{E}(u^\ensddfv)
\leqslant
4 \reg(\ensddfvplat)^2 \normeddfvD{\grad^\ensdiamant u^\ensddfv}^2.
\]
\end{prop}

\begin{demo}
Soient deux mailles primales voisines $\primal{K} \sim \primal{L}$ définissant le diamant $\diamant{D} = \diamant{D}_{\sigma', \dual{\sigma}}$. Par définition du gradient \ref{defi:discrete_gradient} sur $\diamant{D}$, on a
\begin{equation*}
    (u_{\primal{L}} - u_{\primal{K}})^2 
    = \mesure(\dual{\sigma})^2 \abs{\prodscal{\grad^{\diamant{D}} u^\ensddfv}{\vect{\tau}_{\primal{K},\primal{L}}}}^2
    \leqslant \mesure(\dual{\sigma})^2 \abs{\grad^{\diamant{D}} u^\ensddfv}^2.
\end{equation*}
De même, pour deux mailles duales voisines $\mailleduale{K} \sim \mailleduale{L}$ définissant le diamant $\diamant{D} = \diamant{D}_{\sigma,\dual{\sigma}'}$, on a
\begin{equation*}
    (u_{\mailleduale{L}} - u_{\mailleduale{K}})^2 
    = \mesure(\sigma)^2 \abs{\prodscal{\grad^{\diamant{D}} u^\ensddfv}{\vect{\tau}_{\mailleduale{K},\mailleduale{L}}}}^2
    \leqslant \mesure(\sigma)^2 \abs{\grad^{\diamant{D}} u^\ensddfv}^2.
\end{equation*}
Les ensembles $\ensareteprim$ et $\ensaretedual$ sont en bijection avec l'ensemble $\ensdiamant$. Ainsi
\begin{equation} \label{eq:inter}
\mathcal{E}(u^\ensddfv)
\leqslant
\sum_{\diamant{D} \in \ensdiamant} \big[\mesure(\sigma)^2 + \mesure(\dual{\sigma})^2\big] \abs{\grad^{\diamant{D}} u^\ensddfv}^2.
\end{equation}
On souhaite avoir $\mesure(\sigma)^2 + \mesure(\dual{\sigma})^2 \leqslant C \, \poids{\diamant{D}}$ avec $C > 0$ indépendante de $\size(\ensddfvplat)$. On a 
\begin{equation} \label{eq:rapport}
\frac{
    \mesure(\sigma)^2 + \mesure(\dual{\sigma})^2 
}{
    \poids{\diamant{D}}
}
=
\frac{2}{\sin \angle{\diamant{D}}}
\left(
    \frac{\mesure(\sigma)}{\mesure(\dual{\sigma})}
    +
    \frac{\mesure(\dual{\sigma})}{\mesure(\sigma)}
\right).
\end{equation}
Dans la régularité $\reg(\ensddfvplat)$, les termes $\max\limits_{\diamantplat{D}_{\sigma, \dual{\sigma}} \in \ensdiamantplat}\mathopen{}\left[
    \frac{\mesure(\sigma)}{\mesure(\dual{\sigma})},
    \frac{\mesure(\dual{\sigma})}{\mesure(\sigma)}
\right]$ et $\frac{1}{\sin \angle{\ensddfvplat}}$ imposent que l'angle entre deux diagonales ne tende ni vers $0$ ni vers $\pi$ et que le rapport des longueurs des deux diagonales reste uniformément borné. Ainsi, d'après \Cref{eq:rapport}, 
\begin{align*}
\mesure(\sigma)^2 + \mesure(\dual{\sigma})^2 
\leqslant
4 \reg(\ensddfvplat)^2 \poids{\diamant{D}}
\end{align*}
ce qui, avec \ref{eq:inter}, fournit le résultat.
\end{demo}

\noindent Les \Cref{prop:minoration_energie,prop:majoration_energie} aboutissent à une inégalité de type Poincaré mais dont la constante dépend encore de la taille du maillage :
\begin{equation} \label{eq:inegalite_poincare_inter}
\forall u^\ensddfv \in \Rddfvmn, 
\quad
\frac{1}{2 \reg(\ensddfvplat)^2} \, \frac{h(\graphe^\ensddfvplat)}{\size(\ensddfvplat)} \normeddfvT{u^\ensddfv} \leqslant \normeddfvD{\grad^\ensdiamant u^\ensddfv}.
\end{equation}
L'objectif de la section suivante est de s'affranchir de cette dépendance. 

\subsection{Passage du discret au continu}

Dans la suite, nous allons démontrer que sous des hypothèses géométriques appropriées, il existe une constante $c > 0$ indépendante de $\size(\ensddfvplat)$ telle que
\begin{equation} \label{eq:objectif}
h(\graphe^\ensddfvplat) \geqslant c \size(\ensddfvplat).
\end{equation}
Ce résultat, combiné avec \Cref{eq:inegalite_poincare_inter}, fournira le résultat du \Cref{theo:main}.

Pour l'instant, la régularité $\reg(\ensddfvplat)$ contrôle la géométrie locale des mailles et autorise donc les raffinements locaux. Elle ne contrôle donc pas le rapport entre la plus grande et la plus petite taille de maille. Pour montrer \Cref{eq:objectif}, nous allons ajouter une condition de \defini{quasi-uniformité} : il existe $\vartheta > 0$ tel que
\begin{equation} \label{eq:quasi_uniformite}
\vartheta \size(\ensddfvplat) \leqslant \diam{E} \quad \forall E \in \ensprimalplat \cup \ensdualplat \cup \ensdiamantplat.
\end{equation}
Cette hypothèse signifie que les mailles ont toutes une taille de l'ordre de $\size(\ensddfvplat)$. \\

On rappelle que $h(\graphe^\ensddfvplat) = \min\mathopen{}\big\{ h(\graphe^\ensprimal), h(\graphe^\ensdual)\big\}$ (voir \Cref{lemme:ineg_de_graphe}) donc dans l'objectif d'obtenir \ref{eq:objectif}, il va falloir montrer qu'il existe $c^\ensprimal, c^\ensdual > 0$ telles que
\begin{equation} \label{eq:sous_objectif}
h(\graphe^\ensprimal) \geqslant c^\ensprimal \size(\ensddfvplat)
\quad
\text{et}
\quad
h(\graphe^\ensdual) \geqslant c^\ensdual \size(\ensddfvplat).
\end{equation}
Commençons par la première inégalité sur le graphe $\graphe^\ensprimal$. \\
À tout ensemble de sommets $W \subset \vertices(\graphe^\ensprimal)$, on associe une sous-région $\variete{A}_W \subset \surflisse$ consitituée de l'union des mailles primales courbes $\primal{K}$ de $\ensprimal$ dont le représentant $x_{\primal{K}}$ appartient à $W$:
\begin{equation} \label{eq:varieteA_W}
\variete{A}_W \defeq \bigcup_{x_{\primal{K}} \in W} \primal{K}.
\end{equation}
Notre stratégie pour obtenir une inégalité du type \Cref{eq:sous_objectif} consiste à établir deux autres inégalités. Nous allons montrer que pour tout $W \subset \vertices(\graphe^\ensprimal)$,
\begin{equation} \label{eq:inegalites_strategie}
\min\{ \vol(W), \vol(\comp{W}) \} \lesssim \frac{1}{\size(\ensddfvplat)^2} \min\{ \mesure(\variete{A}_W), \mesure(\surflisse \setminus \variete{A}_W) \}
\quad
\text{et}
\quad
\abs{\partial W} \gtrsim \frac{1}{\size(\ensddfvplat)} \mesure(\partial \variete{A}_W).
\end{equation}
Elles permettent d'obtenir
\[
\frac{\abs{\partial W}}{\min\{ \vol(W), \vol(\comp{W}) \}} \gtrsim \size(\ensddfvplat) \frac{\mesure(\partial \variete{A}_W)}{\min\{ \mesure(\variete{A}_W), \mesure(\surflisse \setminus \variete{A}_W) \}} \geqslant \size(\ensddfvplat) \, h(\surflisse),
\]
ce qui, en prenant l'infimum sur $W$, donne
\[
h(\graphe^\ensprimal) \gtrsim \size(\ensddfvplat) \, h(\surflisse).
\]
Comme $\surflisse$ est une surface compacte connexe donnée, sa constante de Cheeger $h(\surflisse)$ est strictement positive (voir \Cref{sec:cheeger_constant}) et nous obtenons alors une inégalité du type \Cref{eq:sous_objectif}. \\

Commençons par établir la proposition suivante selon laquelle les éléments plats et leurs analogues courbes ont essentiellement \chevron{la même taille}.

\begin{lemme} \label{prop:estimations_tailles_plat_vs_courbe}
Il existe des constantes strictement positives $\underline{C}^\ensddfv$, $\overline{C}^\ensddfv$, $\underline{C}^\ensdiamant$ et $\overline{C}^\ensdiamant$ indépendantes de $\size(\ensddfv)$ et dépendantes de $\reg(\ensddfv)$ (refaire la démo de \cite{dziuk_finite_2013} pour bien comprendre de quoi dépendent ces constantes) telles que
\begin{gather}
\forall \primal{K} \in \ensprimal,
\quad
\underline{C}^\ensddfv \mesure(\primalplat{K}) \leqslant \mesure(\primal{K}) \leqslant \overline{C}^\ensddfv \mesure(\primalplat{K}), \label{eq:estimations_tailles_plat_vs_courbe_primal} \\
\forall \mailleduale{K} \in \ensdual,
\quad
\underline{C}^\ensddfv \mesure(\mailledualeplate{K}) \leqslant \mesure(\mailleduale{K}) \leqslant \overline{C}^\ensddfv \mesure(\mailledualeplate{K}), \\
\forall \diamant{D} \in \ensdiamant,
\quad
\underline{C}^\ensdiamant \mesure(\diamantplat{D}) \leqslant \mesure(\diamant{D}) \leqslant \overline{C}^\ensdiamant \mesure(\diamantplat{D}).
\end{gather}
\end{lemme}
Nous avons introduit la condition de quasi-uniformité \ref{eq:quasi_uniformite} pour avoir le résultat suivant. 
\begin{lemme} \label{prop:mino_mesure_primal_K}
Pour tout $\primalplat{K} \in \ensprimalplat$ et $\mailledualeplate{K} \in \ensdualplat$, on a
\[
\frac{\vartheta^2}{\reg(\ensddfvplat)^4} \size(\ensddfvplat)^2 \leqslant \mesure(\primalplat{K})
\quad
\text{et}
\quad
\frac{\vartheta^2}{\reg(\ensddfvplat)^4} \size(\ensddfvplat)^2 \leqslant \mesure(\mailledualeplate{K}),
\]
où $\vartheta$ est la constante de quasi-uniformité.
\end{lemme}

Nous sommes maintenant en mesure de démontrer la proposition suivante fournissant la première inégalité de \Cref{eq:inegalites_strategie}.
\begin{prop} \label{prop:inegalites_strategie_un}
Pour tout $W \subset \vertices(\graphe^\ensprimal)$, on a
\[
\min\{ \vol(W), \vol(\comp{W}) \} \leqslant \frac{\reg(\ensddfvplat)^5}{\underline{C}^\ensddfv \, \vartheta^2 \size(\ensddfvplat)^2} \min\{ \mesure(\variete{A}_W), \mesure(\surflisse \setminus \variete{A}_W) \},
\]
où $\comp{W} = \ens[\big]{v \in \vertices(\graphe^\ensprimal) \tq v \not\in W}$, $\vartheta$ est la constante de quasi-uniformité et $\underline{C}^\ensddfv$ est \areprendre{À compléter}.
\end{prop}
\begin{demo}
Soient $W \subset \vertices(\graphe^\ensprimal)$ et $\variete{A}_W$, la sous-région de $\surflisse$ associée à $W$ et définie par \Cref{eq:varieteA_W}. Puisque les mailles courbes forment une partition de $\variete{A}_W$,
\[
\mesure(\variete{A}_W) = \sum_{x_{\primal{K}} \in W} \mesure(\primal{K}).
\]
Le volume combinatoire de $W$ vaut quant à lui $\vol(W) = \sum\limits_{x_{\primal{K}} \in W} \degr(x_{\primal{K}})$. En appliquant \Cref{eq:estimations_tailles_plat_vs_courbe_primal}, la \Cref{prop:mino_mesure_primal_K} et d'après la \Cref{remarque:NNe_degr}, on a $\degr(x_{\primal{K}}) \leqslant \displaystyle\frac{\reg(\ensddfvplat)^5}{\underline{C}^\ensddfv \, \vartheta^2 \size(\ensddfvplat)^2} \mesure(\primal{K})$ ce qui donne en sommant, 
\[
\vol(W) \leqslant \frac{\reg(\ensddfvplat)^5}{\underline{C}^\ensddfv \, \vartheta^2 \size(\ensddfvplat)^2} \mesure(\variete{A}_W).
\]
De même,
\[
\vol(\comp{W}) \leqslant \frac{\reg(\ensddfvplat)^5}{\underline{C}^\ensddfv \, \vartheta^2 \size(\ensddfvplat)^2} \mesure(\surflisse \setminus \variete{A}_W),
\]
et on en déduit le résultat.
\end{demo}

\noindent Pour démontrer la deuxième inégalité de notre stratégie (\Cref{eq:inegalites_strategie}), nous aurons besoin du résultat suivant qui est l'analogue de \Cref{prop:estimations_tailles_plat_vs_courbe} pour les arêtes.
\begin{lemme} \label{prop:mesure_sigmacourbe}
Il existe $C > 0$ telle que pour toute arête $\sigma \in \ensareteprimplat$,
\[
\mesure(\sigmacourbe) \leqslant \big(1 + C \size(\ensddfvplat)^2\big) \mesure(\sigma).
\]
\end{lemme}
\begin{remarque}
En particulier, comme $\size(\ensddfvplat) < 1$, il existe une constante $C' > 0$ telle que
\[
\mesure(\sigmacourbe) \leqslant (1 + C') \mesure(\sigma). 
\]
\areprendre{$\mesure(\sigmacourbe) \leqslant 2 \mesure(\sigma)$}.
\end{remarque}
La démonstration de \Cref{prop:mesure_sigmacourbe} est également donnée dans \Cref{chap:demo_estimations_geo}. Nous sommes maintenant en mesure de démontrer la proposition suivante fournissant la deuxième inégalité de \Cref{eq:inegalites_strategie}.
\begin{prop}
Il existe $c > 0$ telle que pour tout $W \subset \vertices(\graphe^\ensprimal)$, on a
\[
\abs{\partial W} \geqslant \frac{c}{\size(\ensddfvplat)} \mesure(\partial \variete{A}_W),
\]
où $\partial W$ désigne le \defini{bord} de l'ensemble de sommets $W$ (\ref{eq:partialW}) et $\abs{}$ désigne le nombre l'éléments d'un ensemble.
\end{prop}
\begin{demo}
On a la partition
\[
\partial \variete{A}_W = \bigcup_{\segment{x_{\primalplat{K}}}{x_{\primalplat{L}}} \in \partial W} \projvariete \sigma_{\primalplat{K},\primalplat{L}}
\]
et donc
\[
\mesure(\partial \variete{A}_W) = \sum_{\segment{x_{\primalplat{K}}}{x_{\primalplat{L}}} \in \partial W} \mesure(\projvariete \sigma_{\primalplat{K},\primalplat{L}}).
\]
Or d'après la \Cref{prop:mesure_sigmacourbe}, il existe $C > 0$ telle que
\[
\mesure(\projvariete \sigma_{\primalplat{K},\primalplat{L}})
\leqslant
\big( 1 + C \size(\ensddfvplat)^2 \big) \mesure(\sigma_{\primalplat{K},\primalplat{L}})
\leqslant
\size(\ensddfvplat) \big( 1 + C \size(\ensddfvplat)^2 \big).
\]
Ainsi, 
\[
\mesure(\partial \variete{A}_W) 
\leqslant 
C' \size(\ensddfvplat) \abs{\partial W},
\]
où $C' = \big( 1 + C \size(\ensddfvplat)^2 \big)$. Pour $\size(\ensddfvplat)$ suffisamment petit, on peut prendre $C' = 2$ et on obtient le résultat.
\end{demo}

\section{Construction of the discrete operators}

\cite[Sec.3]{delcourte_discrete_2007} In this section, we approach the gradient, divergence and curl operators by discrete counterparts. We would like to stress that in two dimensions, a distinction is usually made between the vector curl operator from $\R$ to $\R^2$, defined by $\rot u^\ensddfvplat = \displaystyle \Big( \pdv{u^\ensddfvplat}{y}, - \pdv{u^\ensddfvplat}{x} \Big)^\top$ and the scalar curl operator from $\R^2$ to $\R$, defined by $\scarot \xi^\ensdiamantplat = \displaystyle \pdv{\xi^\ensdiamantplat_y}{x} - \pdv{\xi^\ensdiamantplat_x}{y}$.

{\color{red}
Figure \ref{fig:stencils_for_the_discrete_operators} shows the stencils of the different operators and of their combinations. The stencil for the discrete gradient and vector curl operators simply consists of the four \textbf{corners} of the diamond-cell $\diamant{D}$. The stencil for the discrete divergence and scarlar curl operators consists of the diamonds associated to the edges of the primal and dual cells. Arrows are displayed on \ref{fig:stencils_for_the_discrete_operators} to represent the normal and tangential components of the vector fields associated to the diamonds. The stencils for the discrete laplacien on the primal and dual cells respectively consist of the blue and red circles on the figure.
}

\subsection{Construction of the discrete gradient and vector curl operators on the diamond cells}

\areprendre{
We define the discrete gradient of a function $u$ by its values on the {\color{red}diamond-cells of the mesh}. We follow \cite{coudiere_convergence_1999, domelevo_finite_2005} and compute the mean-value of the gradient of any function $u$ on such a cell $\diamant{D}$ by the following formula:
\begin{equation} \label{eq:construction_gradient}
\mesure(\diamant{D}) \mathopen{}\big\langle \grad u_{\restreinta \diamant{D}} \big\rangle = \int_{\diamant{D}} \grad u(x) \d x
= \int_{\partial \diamant{D}} u(\xi) \, \vect{n}(\xi) \d \xi = \sum_{\sigma \in \partial \diamant{D}} \int_\sigma u(\xi) \, \vect{n}(\xi) \d \xi,
\end{equation}
where $\vect{n}(\xi)$ stands for the outward unit normal vector to $\diamant{D}$ at point $\xi$. The integrals in \ref{eq:construction_gradient} can be approximated by the following formula:
\[
\int_{\segment{x_{\primal{K}}}{x_{\mailleduale{K}}}} u(\xi) \d \xi \approx \ell_{\segment{x_{\primal{K}}}{x_{\mailleduale{K}}}} \, \frac{u(x_{\primal{K}}) + u(x_{\mailleduale{K}})}{2},
\]
where $\ell_{\segment{x_{\primal{K}}}{x_{\mailleduale{K}}}}$ denotes the length of the segment $\segment{x_{\primal{K}}}{x_{\mailleduale{K}}}$. Summing the contributions of the different vertices of $\diamant{D}$ and using {\color{myred}elementary geometrical equalities} allows us to give \defini{the definition} of the discrete gradient $\grad^\ensdiamant$ on $\diamant{D}$.
}

\subsubsection{Discrete gradient \texorpdfstring{{\cite{andreianov_discrete_2007,krell_schemas_2010}}}{[ABH07; Kre10]}}

We consider the discrete gradient introduced by Coudière and \emph{al.} in \cite{coudiere_convergence_1999} and applied by Hermeline \cite{hermeline_finite_2000} and Domelevo and Omnès \cite{domelevo_finite_2005} in the framework of \ddfv schemes described above.

\begin{defi}[Discrete gradient] \label{defi:discrete_gradient}
    The \defini{discrete gradient operator} $\fonctionens[\grad^\ensdiamantplat]{\R^\ensddfvplat}{(\R^2)^\ensdiamantplat}$ can be defined as follows: 
    for any $u^\ensddfvplat \in \R^\ensddfvplat$, $\grad^\ensdiamantplat u^\ensddfvplat = (\grad^{\diamantplat{D}} u^\ensddfvplat)_{\diamantplat{D} \in \ensdiamantplat}$ is the function, constant on each diamond cell $\diamantplat{D}_{\sigma, \dual{\sigma}}$, given by
    \begin{equation} \label{eq:discrete_gradient}
    \grad^{\diamantplat{D}} u^\ensddfvplat \cdot \vect{\tau}_{\primalplat{K},\primalplat{L}} = \frac{u_{\primalplat{L}} - u_{\primalplat{K}}}{\mesure(\dual{\sigma})},
    \quad
    \grad^{\diamantplat{D}} u^\ensddfvplat \cdot \vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}} = \frac{u_{\mailledualeplate{L}} - u_{\mailledualeplate{K}}}{\mesure(\sigma)},
    \end{equation}
    where $\diamantplat{D}_{\sigma, \dual{\sigma}}$ is noted $\diamantplat{D}$ when no confusion can arise.
\end{defi}

\begin{remarque}
Le gradient d'une fonction n'est calculé qu'à partir des longueurs des arêtes et des valeurs de la fonction à leurs extrémités.
\end{remarque}
Remark that $\grad^{\diamantplat{D}} u^\ensddfvplat$ can be expressed in the $(\vect{\nu}_{\sigma \primalplat{K}}, \vect{\nu}_{\dual{\sigma} \mailledualeplate{K}})$ basis in the following way:
\begin{equation} \label{eq:discrete_gradient_bis}
\grad^{\diamantplat{D}} u^\ensddfvplat = \frac{1}{\sin \angle{\diamantplat{D}}} \mathopen{}\left( \frac{u_{\primalplat{L}} - u_{\primalplat{K}}}{\mesure(\dual{\sigma})} \, \vect{\nu}_{\sigma \primalplat{K}} + \frac{u_{\mailledualeplate{L}} - u_{\mailledualeplate{K}}}{\mesure(\sigma)} \, \vect{\nu}_{\dual{\sigma} \mailledualeplate{K}} \right)
\end{equation}
thanks to the following lemma.

\begin{lemme}
Using the notations and orientation conventions, for any vector $\xi \in \R^2$ we have
\[
(\sin \angle{\diamantplat{D}}) \, \xi = \prodscal{\xi}{\vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}}} \, \vect{\nu}_{\dual{\sigma} \mailledualeplate{K}} + \prodscal{\xi}{\vect{\tau}_{\primalplat{K},\primalplat{L}}} \, \vect{\nu}_{\sigma \primalplat{K}}.
\]
\end{lemme}

\begin{remarque}
En 2D, d'après la \Cref{prop:mesure_d_un_diamant_plat}, on peut réécrire le gradient discret sous la forme
\begin{equation} \label{eq:gradient_discret_2D}
\grad^{\diamantplat{D}} u^\ensddfvplat = \frac{1}{2\mesure(\diamantplat{D})} \Big[ (u_{\primalplat{L}} - u_{\primalplat{K}}) \mesure(\sigma) \, \vect{\nu}_{\sigma\primalplat{K}} + (u_{\mailledualeplate{L}} - u_{\mailledualeplate{K}}) \mesure(\dual{\sigma}) \, \vect{\nu}_{\dual{\sigma}\mailledualeplate{K}} \Big].
\end{equation}
Note that formula \ref{eq:gradient_discret_2D} is exact for polynomials of degree one. 
\end{remarque}

\subsubsection{Discrete vector curl operator (Gradient tourné)}

\chevron{Le gradient tourné est le tourné du gradient}

\[
\int \grad u \cdot \vect{\tau} \d s = u(1) - u(0)
\]
\[
\int \grad^\perp u \cdot \vect{\nu} \d s = u(1) - u(0)
\]

\begin{defi}[Discrete vector curl operator] \label{defi:discrete_vector_curl_operator}
The \defini{discrete vector curl operator} $\fonctionens[\rot^\ensdiamantplat]{\R^\ensddfvplat}{(\R^2)^\ensdiamantplat}$ can be defined as follows: for any $u^\ensddfvplat \in \R^\ensddfvplat$, $\rot^\ensdiamantplat u^\ensddfvplat = (\rot^{\diamantplat{D}} u^\ensddfvplat)_{\diamantplat{D} \in \ensdiamantplat}$ is the function, constant on each diamond cell $\diamantplat{D}_{\sigma,\dual{\sigma}}$, given by
\begin{equation} \label{eq:discrete_vector_curl}
\prodscal{\rot^{\diamantplat{D}} u^\ensddfvplat}{\vect{\nu}_{\dual{\sigma} \mailledualeplate{K}}} = -\frac{u_{\primalplat{L}} - u_{\primalplat{K}}}{\mesure(\dual{\sigma})},
\quad
\prodscal{\rot^{\diamantplat{D}} u^\ensddfvplat}{\vect{\nu}_{\sigma \primalplat{K}}} = -\frac{u_{\mailledualeplate{L}} - u_{\mailledualeplate{K}}}{\mesure(\sigma)}
\end{equation}
{\color{red}where the unit vectors $\vect{\tau}$ and $\dual{\vect{\tau}}$ are such that $(\vect{\nu},\vect{\tau})$ and $(\dual{\vect{\nu}},\dual{\vect{\tau}})$ are orthonormal positively oriented bases of $\R^2$}. 
\end{defi}
Remark that $\rot^{\diamantplat{D}} u^\ensddfvplat$ can be expressed in the $(\vect{\tau}_{\mailledualeplate{K}, \mailledualeplate{L}}, \vect{\tau}_{\primalplat{K}, \primalplat{L}})$ basis in the following way:
\begin{equation} \label{eq:discrete_vector_curl_bis}
\rot^{\diamantplat{D}} u^\ensddfvplat = -\frac{1}{\sin \angle{\diamantplat{D}}} \mathopen{}\left( \frac{u_{\primalplat{L}} - u_{\primalplat{K}}}{\mesure(\dual{\sigma})} \, \vect{\tau}_{\mailledualeplate{K}, \mailledualeplate{L}} + \frac{u_{\mailledualeplate{L}} - u_{\mailledualeplate{K}}}{\mesure(\sigma)} \, \vect{\tau}_{\primalplat{K}, \primalplat{L}} \right).
\end{equation}

\begin{remarque}[{\cite[Remark 3.3]{delcourte_discrete_2007}}] \label{remarque:fonctionsconstantesoscillations}
In a connected domain, the discrete gradient and vector curl of a given $u^\ensddfvplat \in \R^\ensddfvplat$ vanish if and only if  there exist two constants $c^\ensprimalplat$ and $c^\ensdualplat$, such that 
\[
\forall \primalplat{K} \in \ensprimalplat, \quad u_{\primalplat{K}} = c^\ensprimalplat, 
\qquad
\forall \mailledualeplate{K} \in \ensdualplat, \quad u_{\mailledualeplate{K}} = c^\ensdualplat.
\]
The fact that $c^\ensprimalplat$ and $c^\ensdualplat$ may differ one from the other means that such a $u^\ensddfvplat$ may in general present oscillations.
\end{remarque}

\subsection{Produits scalaires}

\Cref{sec:unknowns} leads us to the definition of the following discrete scalar products. We denote by $\prodscalddfv{}{}$ the inner product on $\R^\ensddfvplat$ given by
\begin{equation} \label{eq:prodscalddfv}
\prodscalddfv{u^\ensddfvplat}{v^\ensddfvplat} \defeq \sum_{\primalplat{K} \in \ensprimalplat} \poids{\primalplat{K}} \, u_{\primalplat{K}} \, v_{\primalplat{K}} + \sum_{\mailledualeplate{K} \in \ensdualplat} \poids{\mailledualeplate{K}} \, u_{\mailledualeplate{K}} \, v_{\mailledualeplate{K}},
\end{equation}
which stands for a discrete $\L^2$ inner product. In the same way, we define a discrete scalar product on the diamond mesh for all $\xi^\ensdiamantplat, \phi^\ensdiamantplat \in (\R^2)^\ensdiamantplat$
\begin{equation} \label{eq:prodscalD}
\prodscalD{\xi^\ensdiamantplat}{\phi^\ensdiamantplat} \defeq \sum_{\diamantplat{D} \in \ensdiamantplat} \poids{\diamantplat{D}} \, \xi^{\diamantplat{D}} \cdot \phi^{\diamantplat{D}}.
\end{equation}
Pour l'instant, nous ne précisons pas la valeur des poids $\poids{\primalplat{K}}$, $\poids{\mailledualeplate{K}}$ et $\poids{\diamantplat{D}}$. Nous y reviendrons ultérieurement. Les deux normes correspondantes sont notées par $\normeddfvT{}$ et $\normeddfvD{}$. Plus généralement, nous posons pour tout $u^\ensddfvplat \in \R^\ensddfvplat$, $\xi^\ensdiamantplat \in (\R^2)^\ensdiamantplat$ et $1 \leqslant p < +\infty$ :
\begin{align} \label{eq:normeddfvTp}
\normeddfvT[p]{u^\ensddfvplat} \defeq \mathopen{}\left(
    \frac{1}{2}  \sum_{\primalplat{K} \in \ensprimalplat} \poids{\primalplat{K}} \abs{u_{\primalplat{K}}}^p +
    \frac{1}{2}  \sum_{\mailledualeplate{K} \in \ensdualplat} \poids{\mailledualeplate{K}} \abs{u_{\mailledualeplate{K}}}^p
\right)^{1/p},
\qquad
&\normeddfvD[p]{\xi^\ensdiamantplat} \defeq
\mathopen{}\left(
    \sum_{\diamantplat{D} \in \ensdiamantplat} \poids{\diamantplat{D}} \abs{\xi^{\diamantplat{D}}}^p
\right)^{1/p}, \\
\normeddfvT[\infty]{u^\ensddfvplat} \defeq \max\mathopen{}\left(
    \max_{\primalplat{K} \in \ensprimalplat} \abs{u_{\primalplat{K}}}, \max_{\mailledualeplate{K} \in \ensdualplat} \abs{u_{\mailledualeplate{K}}}
\right),
\qquad
&\normeddfvD[\infty]{\xi^\ensdiamantplat} \defeq \max_{\diamantplat{D} \in \ensdiamantplat} \abs{\xi^{\diamantplat{D}}}.
\end{align}
Par exemple,
\[
\normeddfvD{\grad^\ensdiamantplat u^\ensddfvplat}^2 = \sum_{\diamantplat{D} \in \ensdiamantplat} \poids{\diamantplat{D}} \abs{\grad^{\diamantplat{D}} u^\ensddfvplat}^2.
\]

\subsubsection{Calcul des poids \texorpdfstring{$\poids{\primalplat{K}}$ et $\poids{\mailledualeplate{K}}$}{muK et muK*}}

Following \Cref{remarque:fonctionsconstantesoscillations}, nous nous placerons dans l'espace des fonctions orthogonales aux gradients nuls, à savoir
\begin{equation} \label{eq:moyenne_nulle}
\Rddfvmn \defeq \R^\ensddfvplat /\, \R = \ens[\bigg]{u^\ensddfvplat \in \R^\ensddfvplat \tq \sum_{\primalplat{K} \in \ensprimalplat} \poids{\primalplat{K}} \, u_{\primalplat{K}} = \sum_{\mailledualeplate{K} \in \ensdualplat} \poids{\mailledualeplate{K}} \, u_{\mailledualeplate{K}} = 0}.
\end{equation}
On prend en général les valeurs
\begin{equation} \label{eq:poidsprimaldual}
\poids{\primalplat{K}} = \frac{\mesure(\primalplat{K})}{2} \quad \text{et} \quad \poids{\mailledualeplate{K}} = \frac{\mesure(\mailledualeplate{K})}{2}.
\end{equation}
On peut prendre d'autres découpages : $\poids{\primalplat{K}} = \alpha \mesure(\primalplat{K})$ et $\poids{\mailledualeplate{K}} = (1 - \alpha) \mesure(\mailledualeplate{K})$ pour $0 < \alpha < 1$. L'essentiel est d'avoir $\normeddfvT{\indicatrice} = \mesure(\surfpoly)$.

\subsubsection{Calcul des poids \texorpdfstring{$\poids{\diamantplat{D}}$}{muD}}

Nous avons introduit un produit scalaire \Cref{eq:prodscalD} sur les fonctions de $(\R^2)^\ensdiamantplat$ de la forme
\[
\forall \xi^\ensdiamantplat, \phi^\ensdiamantplat \in (\R^2)^\ensdiamantplat, \quad
\prodscalD{\xi^\ensdiamantplat}{\phi^\ensdiamantplat} = \sum_{\diamantplat{D} \in \ensdiamantplat} \poids{\diamantplat{D}} \, \xi^{\diamantplat{D}} \cdot \phi^{\diamantplat{D}},
\]
où $(\poids{\diamantplat{D}})_{\diamantplat{D} \in \ensdiamantplat}$ sont des poids à déterminer. On souhaite que ce produit scalaire soit tel qu'on ait un analogue discret de $\rot \perp \grad$. En effet, dans le cas continu, par intégration par parties sur la surface sans bord $\surflisse$ on a
\[
\forall u, v \in \H^1(\surflisse), 
\quad
\int_\surflisse \grad u \cdot \grad^\perp v \d s = 0.
\]
Le but de cette partie est d'obtenir l'analogue discret de cette formule. Calculons alors les poids $(\poids{\diamantplat{D}})_{\diamantplat{D} \in \ensdiamantplat}$ de telle sorte que pour tout $u^\ensddfvplat, v^\ensddfvplat \in \R^\ensddfvplat$, on ait $\prodscalD{\rot^\ensdiamantplat u^\ensddfvplat}{\grad^\ensdiamantplat v^\ensddfvplat} = 0$. D'après les expressions \Cref{eq:discrete_gradient_bis,eq:discrete_vector_curl_bis} et la \Cref{prop:bases_nu_tau}, on a
\[
\prodscalD{\rot^\ensdiamantplat u^\ensddfvplat}{\grad^\ensdiamantplat v^\ensddfvplat}
=
- \sum_{\diamantplat{D} \in \ensdiamantplat} \frac{\poids{\diamantplat{D}}}{\sin(\angle{\diamantplat{D}}) \mesure(\sigma) \mesure(\dual{\sigma})} \, \psi_{\diamantplat{D}}(u^\ensddfvplat, v^\ensddfvplat)
\]
avec $\psi_{\diamantplat{D}}(u^\ensddfvplat, v^\ensddfvplat) = \big(u_{\primalplat{L}} - u_{\primalplat{K}} \big) \big(v_{\mailledualeplate{L}} - v_{\mailledualeplate{K}} \big) + \big(u_{\mailledualeplate{L}} - u_{\mailledualeplate{K}} \big) \big(v_{\primalplat{L}} - v_{\primalplat{K}} \big)$.

Toute fonction $u^\ensddfvplat \in \R^\ensddfvplat$, peut s'écrire sous la forme $u^\ensddfvplat = \sum_{\primalplat{K} \in \ensprimalplat} \indicatrice^\ensddfvplat(\primalplat{K}) \, u_{\primalplat{K}} + \sum_{\mailledualeplate{K} \in \ensdualplat} \indicatrice^\ensddfvplat(\mailledualeplate{K}) \, u_{\mailledualeplate{K}}$. Il y a donc quatre cas à traiter; $(u^\ensddfvplat, v^\ensddfvplat) = \big(\indicatrice^\ensddfvplat(\primalplat{K}), \indicatrice^\ensddfvplat(\primalplat{K})\big)$, $(u^\ensddfvplat, v^\ensddfvplat) = \big(\indicatrice^\ensddfvplat(\mailledualeplate{K}), \indicatrice^\ensddfvplat(\mailledualeplate{K})\big)$, $(u^\ensddfvplat, v^\ensddfvplat) = \big(\indicatrice^\ensddfvplat(\primalplat{K}), \indicatrice^\ensddfvplat(\mailledualeplate{K})\big)$ et $(u^\ensddfvplat, v^\ensddfvplat) = \big(\indicatrice^\ensddfvplat(\mailledualeplate{K}), \indicatrice^\ensddfvplat(\primalplat{K})\big)$. Dans les deux premiers cas, $\psi_{\diamantplat{D}}(u^\ensddfvplat, v^\ensddfvplat) = 0$. Étudions les deux derniers cas pour deux mailles $\primalplat{K} \in \ensprimalplat$ et $\mailledualeplate{K} \in \ensdualplat$ quelconques. On pose $u^\ensddfvplat = \indicatrice^\ensddfvplat(\primalplat{K})$ et $v^\ensddfvplat = \indicatrice^\ensddfvplat(\mailledualeplate{K})$. Si $\ensdiamantplat_{\primalplat{K}} \cap \ensdiamantplat_{\mailledualeplate{K}} = \varnothing$, alors $\psi_{\diamantplat{D}}(u^\ensddfvplat, v^\ensddfvplat) = 0$. Sinon, les mailles $\primalplat{K}$ et $\mailledualeplate{K}$ ont exactement deux diamants en commun et on note $\ensdiamantplat_{\primalplat{K}} \cap \ensdiamantplat_{\mailledualeplate{K}} = \{ \diamantplat{D}_1, \diamantplat{D}_2 \}$. On les représente sur la \Cref{fig:diamants_voisins}. À nouveau, d'après les expressions \Cref{eq:discrete_gradient_bis,eq:discrete_vector_curl_bis}, on a
\begin{align}
    \rot^{\diamantplat{D}_1} u^\ensddfvplat = \frac{1}{\sin \angle{\diamantplat{D}_1}} \frac{-1}{\mesure(\dual{\sigma}_1)} \, \vect{\tau}_{\mailledualeplate{K}, \mailledualeplate{K}_1},
    \quad
    \rot^{\diamantplat{D}_2} u^\ensddfvplat = \frac{1}{\sin \angle{\diamantplat{D}_2}} \frac{1}{\mesure(\dual{\sigma}_2)} \, \vect{\tau}_{\mailledualeplate{K}, \mailledualeplate{K}_2}
\end{align}
et
\begin{align}
    \grad^{\diamantplat{D}_1} v^\ensddfvplat = \frac{1}{\sin \angle{\diamantplat{D}_1}} \frac{-1}{\mesure(\sigma_1)} \, \vect{\nu}_{\dual{\sigma}_1 \mailledualeplate{K}}, \quad
    \grad^{\diamantplat{D}_2} v^\ensddfvplat = \frac{1}{\sin \angle{\diamantplat{D}_2}} \frac{-1}{\mesure(\sigma_2)} \, \vect{\nu}_{\dual{\sigma}_2 \mailledualeplate{K}}
    .
\end{align}
Ainsi,
\[
\prodscalD{\rot^\ensdiamantplat u^\ensddfvplat}{\grad^\ensdiamantplat v^\ensddfvplat}
= 
\frac{\poids{\diamantplat{D}_1}}{\sin(\angle{\diamantplat{D}_1}) \mesure(\sigma_1) \mesure(\dual{\sigma}_1)}
-
\frac{\poids{\diamantplat{D}_2}}{\sin(\angle{\diamantplat{D}_2}) \mesure(\sigma_2) \mesure(\dual{\sigma}_2)}
\]
ce qui impose que pour tout $i \in \{1, 2\}$, $\poids{\diamantplat{D}_i} = \sin(\angle{\diamantplat{D}_i}) \mesure(\sigma_i) \mesure(\dual{\sigma}_i)$. Le calcul pour le cas $(u^\ensddfvplat, v^\ensddfvplat) = \big(\indicatrice^\ensddfvplat(\mailledualeplate{K}), \indicatrice^\ensddfvplat(\primalplat{K})\big)$ donne le même résultat et on décide de multiplier l'expression par $\frac{1}{2}$ de sorte à avoir
\begin{equation} \label{eq:poidsdiamant}
\forall \diamantplat{D} \in \ensdiamantplat, \quad \poids{\diamantplat{D}} = \mesure(\diamantplat{D}).
\end{equation}

\begin{prop} \label{prop:normes}
Soient $u^\ensddfvplat \in \R^\ensddfvplat$ et $\xi^\ensdiamantplat \in (\R^2)^\ensdiamantplat$. 
\begin{itemize}
    \item Normes discrètes :
\begin{align*}
\normeddfvT{u^\ensddfvplat}^2 &= \alpha \sum_{\primalplat{K} \in \ensprimalplat} \mesure(\primalplat{K}) \, u_{\primalplat{K}}{}^2
+ 
(1 - \alpha) \sum_{\mailledualeplate{K} \in \ensdualplat} \mesure(\mailledualeplate{K}) \, u_{\mailledualeplate{K}}{}^2 \\
\normeddfvD{\xi^\ensdiamantplat}^2 &= \sum_{\diamantplat{D} \in \ensdiamantplat} \mesure(\diamantplat{D}) \abs{\xi^{\diamantplat{D}}}^2
\end{align*}
    \item Normes sur la surface lisse $\surflisse$ :
\begin{align*}
\norme{\projddfvvar u^\ensddfvplat}^2_{\L^2(\surflisse)} &=
\frac{1}{2} \sum_{\primal{K} \in \ensprimal} \mesure(\primal{K}) \, u_{\primalplat{K}}{}^2
+
\frac{1}{2} \sum_{\mailleduale{K} \in \ensdual} \mesure(\mailleduale{K}) \, u_{\mailledualeplate{K}}{}^2 \\
\norme{\projdiamantvar \xi^\ensdiamantplat}^2_{\L^2(\surflisse)^2} 
&= \sum_{\diamant{D} \in \ensdiamant} \mesure(\diamant{D}) \abs{\xi^{\diamantplat{D}}}^2
\end{align*}
\end{itemize}
\end{prop}

\begin{remarque}
Nous venons de montrer que les expressions des poids $\poids{\primalplat{K}}$, $\poids{\mailledualeplate{K}}$ et $\poids{\diamantplat{D}}$ (\Cref{eq:poidsprimaldual,eq:poidsdiamant}) sont proportionnelles à la mesure de leur analogue plat. Aussi, d'un point de vue de l'implémentation numérique, calculer la mesure des mailles plates est préférable à calculer la mesure des mailles courbes. Comme le montre la \Cref{prop:normes}, les poids associés aux normes discrètes $\normeddfvT{}$ et $\normeddfvD{}$ sont les mesures des mailles plates et les poids associés aux normes sur la surface lisse $\norme{\projddfvvar \,\cdot\,}^2_{\L^2(\surflisse)}$ et $\norme{\projdiamantvar \,\cdot\,}^2_{\L^2(\surflisse)^2}$ sont les mesures des mailles courbes.
\end{remarque}

\subsection{Écriture des produits scalaires sous forme matricielle}

\begin{envide}[Conventions]
Soient $u^\ensddfvplat \in \R^\ensddfvplat$ et $\xi^\ensdiamantplat \in (\R^2)^\ensdiamantplat$. On note $m = \# \ensprimalplat$, $\dual{m} = \# \ensdualplat$ et $d = \# \ensdiamantplat$. On désigne par
\[
U^\ensddfvplat \defeq \big( u_{\primalplat{K}_1} \quad \cdots \quad u_{\primalplat{K}_m} \quad u_{\mailledualeplate{K}_1} \quad \cdots \quad u_{\mailledualeplate{K}_{\dual{m}}} \big)
\quad
\text{et}
\quad
\Xi^\ensdiamantplat \defeq \big( \xi^{\diamantplat{D}_1} \quad \cdots \quad \xi^{\diamantplat{D}_d} \big)
\]
l'écriture vectorielle de $u^\ensddfvplat$ et de $\xi^\ensdiamantplat$. 
\end{envide}

Le produit scalaire entre deux approximations scalaires $u^\ensddfvplat$ et $v^\ensddfvplat$ peut s'écrire sous forme matricielle:
\[
\prodscalddfv{u^\ensddfvplat}{v^\ensddfvplat} = (U^\ensddfvplat)^\top \masse_\ensddfvplat V^\ensddfvplat
\]
où $\masse_\ensddfvplat$ est la matrice diagonale inversible $\masse_\ensddfvplat \defeq \mathrm{diag}\mathopen{}\big[ (\poids{\primalplat{K}})_{\primalplat{K} \in \ensprimalplat}, (\poids{\mailledualeplate{K}})_{\mailledualeplate{K} \in \ensdualplat} \big]$.

Le produit scalaire entre deux approximations vectorielles $\xi^\ensdiamantplat$ et $\phi^\ensdiamantplat$ peut s'écrire sous forme matricielle également. On note $\Sigma$ la base $(\sigma, \dual{\sigma})$ et $\Xi^{\diamantplat{D}}_\Sigma$ (resp. $\Phi^{\diamantplat{D}}_\Sigma$) l'écriture du vecteur $\xi^{\diamantplat{D}}$ (resp. $\phi^{\diamantplat{D}}$) dans la base $\Sigma$. Par un abus de notation, on écrit $\Xi^\ensdiamantplat_\Sigma = (\Xi^{\diamantplat{D}}_\Sigma)_{\diamantplat{D} \in \ensdiamantplat}$ et de même pour $\Phi^\ensdiamantplat_\Sigma$. On peut alors écrire
\[
\prodscalD{\xi^\ensdiamantplat}{\phi^\ensdiamantplat} = \sum_{\diamantplat{D} \in \ensdiamantplat} (\Xi^{\diamantplat{D}}_\Sigma)^\top \masse^{\diamantplat{D}}_\Sigma \Phi^{\diamantplat{D}}_\Sigma
\]
où
\[
\masse^{\diamantplat{D}}_\Sigma = 
\poids{\diamantplat{D}}
\begin{pmatrix}
    \mesure(\sigma)^2 \, \vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}} \cdot \vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}}
    & \mesure(\sigma) \mesure(\dual{\sigma}) \,
    \vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}} \cdot \vect{\tau}_{\primalplat{K},\primalplat{L}} \\
    \mesure(\sigma) \mesure(\dual{\sigma}) \, \vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}} \cdot \vect{\tau}_{\primalplat{K},\primalplat{L}}
    & \mesure(\dual{\sigma})^2 \, \vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}} \cdot \vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}}
\end{pmatrix}
=
\poids{\diamantplat{D}}
\begin{pmatrix}
    \mesure(\sigma)^2 & \mesure(\sigma) \mesure(\dual{\sigma}) \cos(\angle{\diamantplat{D}}) \\
    \mesure(\sigma) \mesure(\dual{\sigma}) \cos(\angle{\diamantplat{D}})  & \mesure(\dual{\sigma})^2
\end{pmatrix}.
\]
La matrice $\frac{1}{\poids{\diamantplat{D}}} \, \masse^{\diamantplat{D}}_\Sigma$ est la matrice de Gram du produit scalaire dans la base $\Sigma$. On remarque qu'on aurait également pu se placer dans les bases $\big( \vect{\tau}_{\mailledualeplate{K},\mailledualeplate{L}}, \vect{\tau}_{\primalplat{K},\primalplat{L}} \big)$ ou $\big( \vect{\nu}_{\sigma \primalplat{K}}, \vect{\nu}_{\dual{\sigma} \mailledualeplate{K}} \big)$.

\begin{remarque}[HdR Stella]
Pour $u^\ensddfvplat \in \R^\ensddfvplat$ et $\diamantplat{D} \in \ensdiamantplat$, nous définissons $\delta^{\diamantplat{D}} u^\ensddfvplat = \begin{pmatrix}
    u_{\primalplat{K}} - u_{\primalplat{L}} \\ u_{\mailledualeplate{K}} - u_{\mailledualeplate{L}}
\end{pmatrix}$. Puis, pour tout $u^\ensddfvplat, v^\ensddfvplat \in \R^\ensddfvplat$, nous pouvons écrire
\[
\prodscalD{\grad^\ensdiamantplat u^\ensddfvplat}{\grad^\ensdiamantplat v^\ensddfvplat} = 
\sum_{\diamantplat{D} \in \ensdiamantplat} \delta^{\diamantplat{D}} u^\ensddfvplat \cdot \masse_{\diamantplat{D}} \delta^{\diamantplat{D}} v^\ensddfvplat,
\quad
\text{où}
\quad
\masse_{\diamantplat{D}} = 
\frac{1}{4 \, \poids{\diamantplat{D}}{}^2} \masse^{\diamantplat{D}}_\Sigma
\quad
\forall \diamantplat{D} \in \ensdiamantplat.
\]
\end{remarque}

Finalement, en introduisant la matrice symétrique diagonale par blocs $\masse^\ensdiamantplat_\Sigma \defeq \mathrm{diag}\mathopen{}\big[ (\masse^{\diamantplat{D}}_\Sigma)_{\diamantplat{D} \in \ensdiamantplat} \big]$, on peut écrire le produit scalaire vectoriel sous la forme
\begin{align} \label{eq:masseensdiamantSigma}
\prodscalD{\xi^\ensdiamantplat}{\phi^\ensdiamantplat} &= (\Xi^\ensdiamantplat_\Sigma)^\top \masse^\ensdiamantplat_\Sigma \Phi^\ensdiamantplat_\Sigma.
\end{align}
Dans la suite, nous noterons simplement $\masse_\ensdiamantplat = \masse^\ensdiamantplat_\Sigma$. Par exemple, le \Cref{theo:rotperpgrad} se réécrit sous la forme
\[
\forall u^\ensddfvplat, v^\ensddfvplat \in \R^\ensddfvplat, 
\quad
(U^\ensddfvplat)^\top \, (\rotmat_\ensdiamantplat)^\top \masse_\ensdiamantplat \gradmat_\ensdiamantplat V^\ensddfvplat = 0.
\]

\subsubsection{Discrete divergence}

\begin{defi}[Discrete divergence]
L'opérateur de divergence discrète est défini de la manière suivante:
\[
\prodscal{\masse_\ensddfvplat U^\ensddfvplat}{\divmat_\ensddfvplat \Xi^\ensdiamantplat} = - \prodscal{\masse_\ensdiamantplat \gradmat_\ensdiamantplat U^\ensddfvplat}{\Xi^\ensdiamantplat}
\]
\[
\divmat_\ensddfvplat \defeq - \masse^{-1}_\ensddfvplat \gradmat^\top_\ensdiamantplat \masse_\ensdiamantplat
\]
\end{defi}

\begin{prop}
L'opérateur divergence discrète peut s'écrire sous la forme $\fonctionens[\div^\ensddfvplat]{(\R^2)^\ensdiamantplat}{\R^\ensddfvplat}$. Soit $\xi^\ensdiamantplat \in (\R^2)^\ensdiamantplat$, on pose
\[

\div^\ensprimalplat \xi^\ensdiamantplat,
\div^\ensdualplat \xi^\ensdiamantplat    
\big),

avec $\div^\ensprimalplat \xi^\ensdiamantplat = \mathopen{}\big(\div^{\primalplat{K}} \xi^\ensdiamantplat \big)_{\primalplat{K} \in \ensprimalplat}$, $\div^\ensdualplat \xi^\ensdiamantplat = \mathopen{}\big( \div^{\mailledualeplate{K}} \xi^\ensdiamantplat \big)_{\mailledualeplate{K} \in \ensdualplat}$:

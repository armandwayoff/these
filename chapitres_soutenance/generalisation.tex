\subsection{Propriétés des opérateurs discrets}

\begin{frame}{Propriétés des opérateurs discrets}
\begin{theo}[Formules de Green discrètes] \label{theo:discrete_green_formulae}
$\forall \xi^\ensdiamantplat, u^\ensddfvplat \in (\R^2)^\ensdiamantplat \times \R^\ensddfvplat$,
\begin{align}
\prodscalddfv{\div^\ensddfvplat \xi^\ensdiamantplat}{u^\ensddfvplat} &= - \prodscalD{\xi^\ensdiamantplat}{\grad^\ensdiamantplat u^\ensddfvplat} \\
\prodscalddfv{\scarot^\ensddfvplat \xi^\ensdiamantplat}{u^\ensddfvplat} &= \phantom{-} \prodscalD{\xi^\ensdiamantplat}{\rot^\ensdiamantplat u^\ensddfvplat}.
\end{align}
\end{theo}
\begin{align}
    \forall u^\ensddfvplat, v^\ensddfvplat \in \R^\ensddfvplat \\
    \prodscalD{\rot^\ensdiamantplat u^\ensddfvplat}{\grad^\ensdiamantplat v^\ensddfvplat} &= 0 \\
    \div^\ensddfvplat (\rot^\ensdiamantplat u^\ensddfvplat) &= 0 \\
    \scarot^\ensddfvplat (\grad^\ensdiamantplat u^\ensddfvplat) &= 0 \\
    \scarot^\ensddfvplat \rot^\ensdiamantplat u^\ensddfvplat &= - \div^\ensddfvplat \grad^\ensdiamantplat u^\ensddfvplat
\end{align}
\end{frame}

\begin{frame}
\begin{theo}[Décomposition de Helmholtz--Hodge discrète]
\setlength{\abovedisplayskip}{0pt}
\setlength{\abovedisplayshortskip}{0pt}
\begin{gather}
    \forall \xi^\ensdiamantplat \in (\R^2)^\ensdiamantplat,
    \quad
    \exists ! \ (\phi^\ensddfvplat, \psi^\ensddfvplat, h^\ensdiamantplat) \in \Rddfvmn \times \Rddfvmn \times (\R^2)^\ensdiamantplat
    \quad
    \text{t.q.} \\
    \highlightrouge{\xi^\ensdiamantplat = \grad^\ensdiamantplat \phi^\ensddfvplat + \rot^\ensdiamantplat \psi^\ensddfvplat + h^\ensdiamantplat}, \\
    \text{avec $h^\ensdiamantplat \in \param{\mathcal{H}^\ensdiamantplat} \defeq \Ker(\div^\ensddfvplat) \cap \Ker(\scarot^\ensddfvplat)$}
\end{gather}
\end{theo}

\begin{prop}
\[
\dim \param{\mathcal{H}^\ensdiamantplat} = 4 \, g \quad \text{où $g$ est le genre de } \surflisse
\]
\end{prop}
\begin{itemize}
    \item Caractéristique d'Euler : $\chi = S - A + F = 2 - 2 \, g$
    \item $\dim(\R^\ensddfvplat) = S + F$ et $\dim((\R^2)^\ensdiamantplat) = 2 \, A$
\end{itemize}
